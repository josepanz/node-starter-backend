import { ApiProperty } from '@nestjs/swagger';
import { IsOptional, IsString } from 'class-validator';

export class MerchantApplicationRequestDTO {
  @ApiProperty({
    example: 'Acme Corp',
    description: 'Nombre de la empresa solicitante.',
  })
  @IsString()
  @IsOptional()
  merchantName?: string;

  @ApiProperty({
    example: '1345678-1',
    description: 'Número de identificación fiscal de la empresa.',
  })
  @IsString()
  @IsOptional()
  ruc?: string;

  @ApiProperty({
    example: 'Persona Fisica',
    description: 'Tipo de persona del solicitante.',
  })
  @IsString()
  @IsOptional()
  personType?: string;

  @ApiProperty({
    example: 'Razon Social',
    description: 'Razon social de la empresa.',
  })
  @IsString()
  @IsOptional()
  legalName?: string;

  @ApiProperty({
    example: '212312',
    description: 'Codigo actividad economica',
  })
  @IsString()
  @IsOptional()
  businessCode?: string;

  @ApiProperty({
    example: 'Actividad Comercial',
    description: 'Actividad comercial de la empresa.',
  })
  @IsString()
  @IsOptional()
  businessActivity?: string;

  @ApiProperty({
    example: 'Departamento',
    description: 'Departamento donde se encuentra la empresa.',
  })
  @IsString()
  @IsOptional()
  department?: string;

  @ApiProperty({
    example: 'Ciudad',
    description: 'Ciudad donde se encuentra la empresa.',
  })
  @IsString()
  @IsOptional()
  city?: string;

  @ApiProperty({
    example: 'Barrios Altos 123',
    description: 'Barrio en donde se encuentra la empresa.',
  })
  @IsString()
  @IsOptional()
  neighborhood?: string;

  @ApiProperty({
    example: 'Direccion completa',
    description: 'Direccion completa de la empresa',
  })
  @IsString()
  @IsOptional()
  fullAddress?: string;

  @ApiProperty({
    example: '-16.500000',
    description: 'Longitud geográfica de la ubicación de la empresa.',
  })
  @IsString()
  @IsOptional()
  longitude?: string;

  @ApiProperty({
    example: '-68.150000',
    description: 'Latitud geográfica de la ubicación de la empresa.',
  })
  @IsString()
  @IsOptional()
  latitude?: string;

  @ApiProperty({
    example: 'Cedula de Identidad',
    description: 'Tipo de documento del representante legal.',
  })
  @IsString()
  @IsOptional()
  documentType?: string;

  @ApiProperty({
    example: '12345678',
    description: 'Número de documento del representante legal.',
  })
  @IsString()
  @IsOptional()
  documentNumber?: string;

  @ApiProperty({
    example: 'URL',
    description: 'URL de la imagen frontal del documento de identidad.',
  })
  @IsString()
  @IsOptional()
  documentFrontImage?: string;

  @ApiProperty({
    example: 'URL',
    description: 'URL de la imagen posterior del documento de identidad.',
  })
  @IsString()
  @IsOptional()
  documentBackImage?: string;

  @ApiProperty({
    example: 'URL',
    description: 'URL de la imagen de factura comercial',
  })
  @IsString()
  @IsOptional()
  commercialInvoiceImage?: string;

  @ApiProperty({
    example: 'URL',
    description: 'URL de la imagen de la patente comercial',
  })
  @IsString()
  @IsOptional()
  businessLicenseImage?: string;

  @ApiProperty({
    example: 'URL',
    description: 'URL de la imagen de la fachada del comercio',
  })
  @IsString()
  @IsOptional()
  storefrontImage?: string;

  @ApiProperty({
    example: 'Juan Perez',
    description: 'Nombre del titular de la cuenta.',
  })
  @IsString()
  @IsOptional()
  bankAccountHolderName?: string;

  @ApiProperty({
    example: 'RUC del titular',
    description: 'RUC del titular de la cuenta bancaria.',
  })
  @IsString()
  @IsOptional()
  accountHolderRuc?: string;

  @ApiProperty({
    example: 'Banco de Ejemplo',
    description: 'Nombre del banco donde se tiene la cuenta.',
  })
  @IsString()
  @IsOptional()
  bankName?: string;

  @ApiProperty({
    example: 'Cuenta de Ahorros',
    description:
      'Tipo de cuenta bancaria (ej. Cuenta de Ahorros, Cuenta Corriente).',
  })
  @IsString()
  @IsOptional()
  accountType?: string;

  @ApiProperty({
    example: '1234567890',
    description: 'Número de cuenta bancaria.',
  })
  @IsString()
  @IsOptional()
  bankAccountNumber?: string;
}
