import { MerchantApplicationRequestDTO } from '@api/onboarding/dto';
import {
  BankData,
  CommercialData,
  DocumentsData,
  LocationData,
} from '../interfaces/merchants-application.interface';

export class ValidationDataHelper {
  static validateCommercialData(
    dto: MerchantApplicationRequestDTO,
  ): CommercialData | null {
    const data = {
      merchantName: dto.merchantName?.trim() || null,
      ruc: dto.ruc?.trim() || null,
      personType: dto.personType?.trim() || null,
      legalName: dto.legalName?.trim() || null,
      merchantCode: dto.businessCode?.trim() || null,
      businessActivity: dto.businessActivity?.trim() || null,
    };

    const isComplete = Object.values(data).every(
      (value) => value !== null && value !== '',
    );

    if (!isComplete) return null;

    return {
      merchantName: data.merchantName!,
      ruc: data.ruc!,
      personType: data.personType!,
      legalName: data.legalName!,
      businessActivity: data.businessActivity!,
    };
  }

  }

  static normalizeStrings<T extends Record<string, string | null>>(
    obj: T,
  ): { [K in keyof T]?: string } {
    const result: Partial<Record<keyof T, string>> = {};

    for (const key in obj) {
      const value = obj[key];

      if (value === null || value === undefined) continue;

      result[key] = value.trim();
    }

    return result as { [K in keyof T]?: string };
  }
