export class StepDTO {
  @ApiProperty({
    example: 'COMMERCIAL_DATA',
    description: 'Identificador único del paso',
  })
  @IsString()
  id!: string;

  @ApiProperty({
    description: 'Datos del paso. Puede variar según el id.',
    oneOf: [
      { $ref: '#/components/schemas/CommercialDataDTO' },
      { $ref: '#/components/schemas/LocationDataDTO' },
      { $ref: '#/components/schemas/DocumentsDataDTO' },
      { $ref: '#/components/schemas/BankDataDTO' },
    ],
    nullable: true,
  })
  @IsOptional()
  @ValidateNested()
  data?:
    | CommercialDataDTO
    | LocationDataDTO
    | DocumentsDataDTO
    | BankDataDTO
    | null;
}

export class MerchantApplicationResponseDTO {
  @ApiProperty({
    example: 'DRAFT',
    enum: ['DRAFT', 'PENDING', 'APPROVED', 'REJECTED'],
    nullable: true,
  })
  @IsString()
  @IsOptional()
  status?: string | null;

  @ApiProperty({
    type: [StepDTO],
    description: 'Lista de pasos de la solicitud',
  })
  @IsArray()
  @ValidateNested({ each: true })
  @Type(() => StepDTO)
  steps?: StepDTO[];

  @ApiProperty({
    example: 'DOCUMENTS',
    description: 'ID del paso actual',
    nullable: true,
  })
  @IsString()
  @IsOptional()
  currentStep?: string | null;

  @ApiProperty({ example: 2, description: 'Cantidad de pasos ya completados' })
  @IsNumber()
  completedSteps!: number;

  @ApiProperty({ example: 4, description: 'Total de pasos configurados' })
  @IsNumber()
  totalSteps!: number;
}
